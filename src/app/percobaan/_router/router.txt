import { z } from "zod";
import V2ClientRouter from "../_lib/ClientRouter";

const dashboard = new V2ClientRouter({
  prefix: "/dashboard",
  name: "dashboard",
})
  .add("/", {
    query: z.object({
      page: z.string(),
    }),
  })
  .add("/berita");

const router = new V2ClientRouter({
  prefix: "/percobaan",
  name: "percobaan",
}).use("dashboard", dashboard);

export default router;
